SQLtestada
-- retencao_12m · v2.1 · lê o contrato clientes_pagantes
select count(distinct c.conta_id) filter (where c.pagante_hoje)
       / nullif(count(distinct c.conta_id), 0) as retencao_12m
from contratos.clientes_pagantes c
where c.pagante_em = current_date - interval '12 months'
  and not c.conta_de_teste
